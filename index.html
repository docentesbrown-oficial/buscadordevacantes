Revisá el nombre de la pestaña en Google Sheets.'); } const lastRow = sheet.getLastRow(); if (lastRow < CONFIG.FIRST_DATA_ROW) { return { records: [], count: 0, updatedAt: new Date().toISOString() }; } const numberOfRows = lastRow - CONFIG.HEADER_ROW; // LISTA BLANCA DE COLUMNAS: // A:D = código escuela, curso/sección, PID, CUPOF. // F = nivel-modalidad. // L = fecha de producción de la vacante. // W:AC = módulo/cargo, cantidad de módulos, lunes a viernes. // Se hacen lecturas separadas para NO tocar las columnas intermedias ocultas. const colsAtoD = sheet.getRange(CONFIG.FIRST_DATA_ROW, 1, numberOfRows, 4).getDisplayValues(); const colF = sheet.getRange(CONFIG.FIRST_DATA_ROW, 6, numberOfRows, 1).getDisplayValues(); const colL = sheet.getRange(CONFIG.FIRST_DATA_ROW, 12, numberOfRows, 1).getDisplayValues(); const colsWtoAC = sheet.getRange(CONFIG.FIRST_DATA_ROW, 23, numberOfRows, 7).getDisplayValues(); const records = []; for (let i = 0; i < numberOfRows; i++) { const base = colsAtoD[i]; const services = colsWtoAC[i]; const record = { id: 'fila-' + (CONFIG.FIRST_DATA_ROW + i), escuelaCodigo: clean_(base[0]), // A cursoSeccion: clean_(base[1]), // B pid: clean_(base[2]), // C cupof: clean_(base[3]), // D nivelModalidad: clean_(colF[i][0]), // F fechaProduceVacante: clean_(colL[i][0]), // L prestacion: clean_(services[0]), // W cantidadModulos: clean_(services[1]), // X dias: [] }; for (let dayIndex = 0; dayIndex < CONFIG.DAY_NAMES.length; dayIndex++) { const horario = clean_(services[dayIndex + 2]); // Y:AC if (horario) { record.dias.push({ dia: CONFIG.DAY_NAMES[dayIndex], horario: horario }); } } // Omite únicamente filas sin ningún dato en las columnas autorizadas. const hasContent = [ record.escuelaCodigo, record.cursoSeccion, record.pid, record.cupof, record.nivelModalidad, record.fechaProduceVacante, record.prestacion, record.cantidadModulos ].some(Boolean) || record.dias.length > 0; if (hasContent) records.push(record); } return { records: records, count: records.length, updatedAt: new Date().toISOString() }; } /** Convierte valores a texto sin exponer datos adicionales. */ function clean_(value) { return value === null || value === undefined ? '' : String(value).trim(); }
